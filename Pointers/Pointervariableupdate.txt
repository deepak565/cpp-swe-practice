// pointers increment and decrement

#include <iostream>
using namespace std;

void IncrementorDecrement(int *x, float *y, char *z) {
    *x = *x - 1 ;
    *x = *x + 100;
    
    *y = *y - 1.2;
    *y = *y + 100.9222;
    
    *z = *z + 1;
    *z = *z - 11;
}


int main() {
int x = 10;
float y = 10.5;
char z = 'a';

IncrementorDecrement(&x, &y, &z);

cout<<"value of x "<<x<<endl;
cout<<"value of y "<<y<<endl;
cout<<"value of z "<<z<<endl;
    
    
    return 0;
}
