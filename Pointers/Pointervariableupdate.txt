// call by reference
#include <iostream>

void doubleValue(int &x) {
    x = x * 2;
}

int main() {
    int x = 10;              // Correct: declaring an int variable
    doubleValue(x);          // Pass by reference
    std::cout << "Doubled value: " << x << std::endl;

    return 0;
}
