// PROGRAM:
#include <iostream>
#include <thread>
#include <chrono>
#include <algorithm>
#include <future>

using namespace std;
using namespace std::chrono;
typedef long int ull;

void findOdd(std::promise<ull>&& OddSumPromise, ull start, ull end) {
    ull OddSum = 0;
    for (ull i = start; i <= end; ++i) {
        if (i % 2 != 0) {
            OddSum += i;
        }
    }
    OddSumPromise.set_value(OddSum);
}
